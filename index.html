Quick Start Commands
# Convert PWA to Android/iOS with Capacitor
npm install @capacitor/core @capacitor/cli
npx cap init SmartPlateAI com.smartplate.ai
npx cap add android
npx cap add ios
npx cap sync

# Convert to Desktop with Electron
npm install electron --save-dev
# Create main.js and package.json
npm run make

# Generate all formats with PWA Builder
npx @pwabuilder/pwabuilder build
